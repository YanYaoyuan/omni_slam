cmake_minimum_required(VERSION 3.16)
project(fast_livo)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(std_srvs REQUIRED)
find_package(omni_slam_interfaces REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(visualization_msgs REQUIRED)
find_package(tf2_ros REQUIRED)
find_package(pcl_conversions REQUIRED)
find_package(cv_bridge REQUIRED)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED COMPONENTS common io filters)
find_package(OpenCV REQUIRED)
find_package(OpenMP REQUIRED)
option(FAST_LIVO_USE_LIVOX "Enable livox_ros_driver2 CustomMsg input" OFF)
set(ros_dependencies rclcpp std_srvs omni_slam_interfaces sensor_msgs nav_msgs geometry_msgs visualization_msgs tf2_ros pcl_conversions cv_bridge)
if(FAST_LIVO_USE_LIVOX)
  find_package(livox_ros_driver2 REQUIRED)
  list(APPEND ros_dependencies livox_ros_driver2)
  add_compile_definitions(FAST_LIVO_USE_LIVOX)
endif()
# Build pinned upstream dependencies without catkin or machine-wide installs.
set(sophus_dir ${CMAKE_CURRENT_SOURCE_DIR}/third_party/Sophus)
set(vikit_dir ${CMAKE_CURRENT_SOURCE_DIR}/third_party/rpg_vikit/vikit_common)
add_library(fast_livo_sophus STATIC ${sophus_dir}/sophus/so2.cpp ${sophus_dir}/sophus/so3.cpp ${sophus_dir}/sophus/se3.cpp)
target_include_directories(fast_livo_sophus PUBLIC ${sophus_dir})
target_link_libraries(fast_livo_sophus PUBLIC Eigen3::Eigen)
add_library(fast_livo_vikit STATIC ${vikit_dir}/src/math_utils.cpp ${vikit_dir}/src/vision.cpp ${vikit_dir}/src/pinhole_camera.cpp ${vikit_dir}/src/robust_cost.cpp ${vikit_dir}/src/performance_monitor.cpp)
target_include_directories(fast_livo_vikit PUBLIC ${vikit_dir}/include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(fast_livo_vikit PUBLIC fast_livo_sophus ${OpenCV_LIBS})
add_library(fast_livo_core STATIC src/vio.cpp src/frame.cpp src/visual_point.cpp src/voxel_map.cpp src/preprocess.cpp src/IMU_Processing.cpp src/LIVMapper.cpp)
target_include_directories(fast_livo_core PUBLIC include ${PCL_INCLUDE_DIRS})
target_compile_definitions(fast_livo_core PRIVATE MP_EN MP_PROC_NUM=3)
target_link_libraries(fast_livo_core fast_livo_vikit ${PCL_LIBRARIES} OpenMP::OpenMP_CXX)
ament_target_dependencies(fast_livo_core ${ros_dependencies})
add_executable(fastlivo_mapping src/main.cpp)
target_link_libraries(fastlivo_mapping fast_livo_core)
ament_target_dependencies(fastlivo_mapping ${ros_dependencies})
add_executable(fastlivo_offline src/offline_main.cpp)
target_link_libraries(fastlivo_offline fast_livo_core)
ament_target_dependencies(fastlivo_offline ${ros_dependencies})
install(TARGETS fastlivo_mapping fastlivo_offline DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
install(FILES LICENSE UPSTREAM.md DESTINATION share/${PROJECT_NAME})
install(PROGRAMS scripts/prepare_vbot_bag.py scripts/run_bag_experiment.py scripts/analyze_experiment.py scripts/inspect_full_bags.py scripts/run_full_bags.py scripts/analyze_full_bags.py scripts/audit_vbot_projection.py DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/optimize_keyframe_map.py scripts/audit_map_quality.py scripts/extract_revisit_images.py DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/prepare_managed_vbot.py scripts/build_semantic_map.py DESTINATION lib/${PROJECT_NAME})
install(FILES scripts/revisit_geometry.py scripts/map_quality.py DESTINATION lib/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_vanjee_preprocess test/test_vanjee_preprocess.cpp)
  target_link_libraries(test_vanjee_preprocess fast_livo_core)
endif()
ament_package()
